import { launch, pumpSheet, helpers } from './harness.mjs'
const OUT = process.env.OUT
const { tab, start, close, errors } = await launch({ width: 1900, height: 1000 })
const log = (k, v) => console.log(k + ':', JSON.stringify(v))
try {
  await start(await pumpSheet([
    { id: 'locked', type: 'rectangle', x: 700, y: 120, width: 160, height: 60, locked: true, customData: { kind: 'math', source: 'p = 3 bar\nq = p * 2' } },
    { id: 'box', type: 'rectangle', x: 400, y: 120, width: 120, height: 80, strokeColor: '#1e1e1e', backgroundColor: 'transparent' },
  ]))
  const h = await helpers(tab)
  await tab.evaluate(() => window.__api.updateScene({ appState: { scrollX: 60, scrollY: 80, zoom: { value: 1 } } })); await tab.waitForTimeout(300)
  log('elements', (await h.els()).map((e) => ({ id: e.id, x: e.x, y: e.y, w: e.w, h: e.h, locked: e.locked })))
  // ---- Lasso: a loop around the small element, the box and the locked element (scene x 20..900, y 90..240)
  await h.moreTool('Lasso selection')
  log('tool', await h.state(['activeTool']))
  await h.drag([[20, 90], [300, 80], [600, 85], [900, 90], [905, 170], [900, 245], [600, 250], [300, 248], [20, 245], [15, 170], [20, 92]], { steps: 10 })
  await tab.waitForTimeout(500)
  log('lasso around calc, box and the locked element', await h.state(['selectedElementIds', 'activeTool']))
  await tab.screenshot({ path: OUT + '/06-lasso.png' })
  log('hint', await tab.evaluate(() => { const v = document.querySelector('.HintViewer'); return v ? { text: v.textContent, after: getComputedStyle(v.querySelector('span') ?? v, '::after').content } : null }))
  // Enter with several selected; then a lasso around the math element alone
  await tab.keyboard.press('Escape'); await tab.waitForTimeout(200)
  log('after Esc', await h.state(['selectedElementIds', 'activeTool']))
  if ((await h.state(['activeTool'])).activeTool !== 'lasso') await h.moreTool('Lasso selection')
  await h.drag([[20, 100], [150, 95], [270, 100], [275, 170], [270, 235], [150, 240], [20, 235], [15, 170], [20, 102]], { steps: 10 })
  await tab.waitForTimeout(500)
  log('lasso around the math element alone', await h.state(['selectedElementIds', 'activeTool']))
  log('hint with one math element lassoed', await tab.evaluate(() => { const v = document.querySelector('.HintViewer'); const s = v?.querySelector('span'); return v ? { text: v.textContent, after: s ? getComputedStyle(s, '::after').content : null } : null }))
  log('properties panel (reduced for math?)', await tab.evaluate(() => [...document.querySelectorAll('.App-menu__left .control-label, .App-menu__left legend, .App-menu__left h3')].filter((e) => e.offsetParent !== null).map((e) => e.textContent.trim().slice(0, 30))))
  await tab.keyboard.press('Enter'); await tab.waitForTimeout(500)
  log('Enter on the lassoed math element opens', { line: await tab.evaluate(() => document.querySelector('.math-element__field')?.value ?? null) })
  await tab.keyboard.press('Escape'); await tab.waitForTimeout(300)
  const r1 = await tab.locator('[data-anchor-id="calc"] [data-line="1"]').boundingBox()
  await tab.mouse.dblclick(r1.x + 20, r1.y + 12); await tab.waitForTimeout(500)
  log('double-click a line with the lasso tool active', { line: await tab.evaluate(() => document.querySelector('.math-element__field')?.value ?? null), ...(await h.state(['activeTool', 'selectedElementIds'])) })
  await tab.keyboard.press('Escape'); await tab.keyboard.press('Escape'); await tab.waitForTimeout(300)
  // a lasso drag that starts inside the math element: does it move the element or lasso?
  const before = (await h.els()).find((e) => e.id === 'calc')
  if ((await h.state(['activeTool'])).activeTool !== 'lasso') await h.moreTool('Lasso selection')
  await h.drag([[100, 160], [130, 200], [200, 260], [260, 300]], { steps: 10 })
  await tab.waitForTimeout(400)
  const after = (await h.els()).find((e) => e.id === 'calc')
  log('a lasso drag begun inside the math element', { moved: before.x !== after.x || before.y !== after.y, from: [before.x, before.y], to: [after.x, after.y], ...(await h.state(['selectedElementIds'])) })
  await tab.keyboard.press('Control+z'); await tab.waitForTimeout(300)

  // ---- Tab on a selected anchor
  await tab.keyboard.press('Escape'); await h.focusCanvas(); await tab.keyboard.press('v'); await tab.waitForTimeout(200)
  const r0 = await tab.locator('[data-anchor-id="calc"] [data-line="0"]').boundingBox()
  await tab.mouse.click(r0.x + 20, r0.y + 12); await tab.waitForTimeout(300)
  log('selected', await h.state(['selectedElementIds']))
  await tab.keyboard.press('Tab'); await tab.waitForTimeout(400)
  log('after one Tab', { popup: await tab.evaluate(() => [...document.querySelectorAll('[class*=ConvertElementTypePopup], [class*=convert], .ConvertElementTypePopup')].map((e) => e.className).slice(0, 3)), type: (await h.els()).find((e) => e.id === 'calc')?.type, mathDrawn: await tab.evaluate(() => !!document.querySelector('[data-anchor-id="calc"]')), focus: await tab.evaluate(() => document.activeElement?.tagName + '.' + String(document.activeElement?.className).slice(0, 60)) })
  await tab.screenshot({ path: OUT + '/06-tab-1.png' })
  await tab.keyboard.press('Tab'); await tab.waitForTimeout(500)
  const el = (await h.els()).find((e) => e.id === 'calc')
  log('after a second Tab', { type: el?.type, math: el?.math, mathDrawn: await tab.evaluate(() => !!document.querySelector('[data-anchor-id="calc"]')), variables: await tab.evaluate(() => [...document.querySelectorAll('.sheet-sidebar [class*=group-title], .sheet-sidebar h3, .sheet-sidebar [class*=element]')].map((e) => e.textContent).slice(0, 4)) })
  await tab.screenshot({ path: OUT + '/06-tab-2.png' })
  await tab.keyboard.press('Escape'); await tab.waitForTimeout(300)
  const file = JSON.parse(await tab.evaluate(() => window.__leaf.view.getViewData()))
  log('the converted anchor in the file', file.elements.filter((e) => e.id === 'calc').map((e) => ({ type: e.type, customData: e.customData, bg: e.backgroundColor, strokeStyle: e.strokeStyle })))
  await h.focusCanvas(); await tab.keyboard.press('Control+z'); await tab.waitForTimeout(500)
  const back = (await h.els()).find((e) => e.id === 'calc')
  log('after Ctrl+Z', { type: back?.type, mathDrawn: await tab.evaluate(() => !!document.querySelector('[data-anchor-id="calc"]')) })
} finally { log('ERRORS', errors); await close() }
