import { readFile } from 'node:fs/promises'
import { launch } from './harness.mjs'
const OUT = process.env.OUT
const { tab, start, close, errors } = await launch({ width: 1900, height: 1000 })
try {
  const saved = await readFile(OUT + '/../fill.excalimathpad', 'utf8')
  console.log('saved by the canary:', JSON.stringify(JSON.parse(saved).elements.map((e) => e.type + (e.polygon ? '(polygon)' : '') + (e.customData?.kind === 'math' ? '(math)' : ''))), JSON.stringify(JSON.parse(saved).appState))
  await start(saved)
  await tab.screenshot({ path: OUT + '/11-fill-in-041.png' })
  const row = await tab.locator('[data-anchor-id="pump"] [data-line="3"]').boundingBox()
  await tab.mouse.click(row.x + 30, row.y + 12); await tab.waitForTimeout(800); await tab.mouse.dblclick(row.x + 30, row.y + 12); await tab.waitForTimeout(500)
  await tab.keyboard.press('End'); await tab.keyboard.type(' '); await tab.keyboard.press('Escape'); await tab.waitForTimeout(3500)
  const disk = await tab.evaluate(async (path) => { const app = window.__leaf.view.app; const f = app.vault.getAbstractFileByPath?.(path) ?? app.vault.getFileByPath?.(path); return app.vault.read(f) }, 'Calcs/pump.excalimathpad')
  const d = JSON.parse(disk)
  console.log('after an edit in 0.4.1:', JSON.stringify(d.elements.map((e) => e.type + (e.polygon ? '(polygon)' : '') + (e.backgroundColor && e.type === 'line' ? ' ' + e.backgroundColor : ''))), JSON.stringify(d.appState))
} finally { console.log(JSON.stringify(errors)); await close() }
