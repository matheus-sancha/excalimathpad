// Which lines of a three-line math element a double-click opens: the canary against 0.4.1 (DOM only, so it runs on both).
import { launch, pumpSheet } from './harness.mjs'
const { tab, start, close, errors } = await launch({ width: 1900, height: 1000 })
try {
  await start(await pumpSheet())
  const out = []
  for (const [line, dx] of [[0, 20], [1, 20], [2, 20], [1, 5], [1, 120], [1, 160]]) {
    const r = await tab.locator(`[data-anchor-id="calc"] [data-line="${line}"]`).boundingBox()
    const el = await tab.locator('[data-anchor-id="calc"]').boundingBox()
    await tab.mouse.click(r.x + dx, r.y + 12); await tab.waitForTimeout(800)
    await tab.mouse.dblclick(r.x + dx, r.y + 12); await tab.waitForTimeout(600)
    out.push({ line, dx, fromCentre: [Math.round(r.x + dx - (el.x + el.width / 2)), Math.round(r.y + 12 - (el.y + el.height / 2))], opened: await tab.evaluate(() => document.querySelector('.math-element__field')?.value ?? null), textEditor: await tab.evaluate(() => !!document.querySelector('.excalidraw-wysiwyg')) })
    await tab.keyboard.press('Escape'); await tab.keyboard.press('Escape'); await tab.waitForTimeout(300)
  }
  for (const o of out) console.log(JSON.stringify(o))
  // Enter on the selected element
  const r = await tab.locator('[data-anchor-id="calc"] [data-line="1"]').boundingBox()
  await tab.mouse.click(r.x + 20, r.y + 12); await tab.waitForTimeout(500); await tab.keyboard.press('Enter'); await tab.waitForTimeout(500)
  console.log('Enter on the selected element opens', await tab.evaluate(() => document.querySelector('.math-element__field')?.value ?? null), 'text editor:', await tab.evaluate(() => !!document.querySelector('.excalidraw-wysiwyg')))
} finally { console.log(JSON.stringify(errors)); await close() }
