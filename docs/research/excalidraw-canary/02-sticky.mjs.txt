import { writeFile } from 'node:fs/promises'
import { launch, pumpSheet, helpers } from './harness.mjs'
const OUT = process.env.OUT
const { tab, start, close, errors } = await launch()
const log = (k, v) => console.log(k + ':', JSON.stringify(v))
try {
  await start(await pumpSheet())
  const h = await helpers(tab)
  log('elements at start', await h.els())
  await h.focusCanvas()
  await tab.keyboard.press('n')
  log('after N', await h.state(['activeTool']))
  await h.click(900, 140)
  await tab.waitForTimeout(400)
  log('after click on empty canvas', { ...(await h.state(['activeTool', 'selectedElementIds'])), editing: await tab.evaluate(() => !!window.__api.getAppState().editingTextElement), active: await tab.evaluate(() => document.activeElement?.tagName + '.' + document.activeElement?.className) })
  await tab.keyboard.type('note beside')
  await tab.keyboard.press('Escape')
  await tab.waitForTimeout(300)
  log('elements after first note', (await h.els()).filter((e) => !e.math))
  // A note on a math element: the tool, then a click inside the small element.
  await h.focusCanvas()
  await tab.keyboard.press('n')
  await h.click(60, 150)
  await tab.waitForTimeout(400)
  await tab.keyboard.type('note over math')
  await tab.keyboard.press('Escape')
  await tab.waitForTimeout(400)
  log('elements after note over math', (await h.els()).filter((e) => !e.math))
  log('order in scene', (await h.els()).map((e) => e.id.slice(0, 6) + ':' + e.type))
  await tab.keyboard.press('Escape')
  await tab.waitForTimeout(300)
  await tab.screenshot({ path: OUT + '/02-sticky.png' })
  // Does a press on the note-over-math area still reach math (double-click a line under the note)?
  const row = await tab.locator('[data-anchor-id="calc"] [data-line="1"]').boundingBox()
  await tab.mouse.dblclick(row.x + 20, row.y + row.height / 2)
  await tab.waitForTimeout(500)
  log('double-click on a math line under the note', { lineOpen: await tab.evaluate(() => !!document.querySelector('.math-element__field')), ...(await h.state(['selectedElementIds'])), editingText: await tab.evaluate(() => !!window.__api.getAppState().editingTextElement) })
  await tab.keyboard.press('Escape'); await tab.keyboard.press('Escape')
  const file = await tab.evaluate(() => window.__leaf.view.getViewData())
  const parsed = JSON.parse(file)
  log('file: element types', parsed.elements.map((e) => e.type))
  log('file: a sticky note as saved', parsed.elements.find((e) => e.type !== 'rectangle' && e.type !== 'text'))
  log('file: its text, if a separate element', parsed.elements.filter((e) => e.type === 'text'))
  log('file: block', parsed.excalimathpad)
  await writeFile(OUT + '/../sticky.excalimathpad', file)
} finally { log('ERRORS', errors); await close() }
