+ * line is committed, or a Rename command?"

+ * Three variants of how a rename starts, on the pump sheet, in the real app,
+ * switchable with `?rename=A|B|C` and the bar at the bottom:

+// ───────────────────────────── the rename, prototype grade ─────────────────────────────
+
+const ID_START = /[A-Za-z_$À-ʯͰ-Ͽ]/
+const ID_CHAR = /[A-Za-z0-9_$À-ʯͰ-Ͽ]/
+
+interface Token {
+  start: number
+  end: number
+  name: string
+}
+
+/** The names in a line's code: not in a text, a note, a number or after `.` or `@`. */
+function names(line: string): Token[] {
+  const found: Token[] = []
+  let i = 0
+  while (i < line.length) {
+    const char = line[i]
+    if (char === '#') break
+    if (char === '"') {
+      const close = line.indexOf('"', i + 1)
+      i = close < 0 ? line.length : close + 1
+    } else if (/\d/.test(char)) {
+      i += /^\d*\.?\d*(e[+-]?\d+)?/i.exec(line.slice(i))![0].length || 1
+    } else if (ID_START.test(char)) {
+      let end = i + 1
+      while (end < line.length && ID_CHAR.test(line[end])) end++
+      const before = line[i - 1]
+      if (before !== '.' && before !== '@') found.push({ start: i, end, name: line.slice(i, end) })
+      i = end
+    } else i++
+  }
+  return found
+}
+
+/** What a line assigns: `rho = …` or `v(Q) = …`. */
+function definition(line: string): { name: string; params: string[] } | null {
+  const match = /^\s*([A-Za-z_$À-ʯͰ-Ͽ][A-Za-z0-9_$À-ʯͰ-Ͽ]*)\s*(?:\(([^)]*)\))?\s*=(?!=)/.exec(line)
+  if (!match) return null
+  return { name: match[1], params: match[2] === undefined ? [] : match[2].split(',').map((param) => param.trim()) }
+}

+interface SheetElement {
+  anchorId: string
+  label: string
+  lines: string[]
+}
+
+function readSheet(api: ExcalidrawImperativeAPI): SheetElement[] {
+  const anchors = api.getSceneElements().filter(isMathAnchor).map(toMathAnchor)
+  return readingOrder(anchors).map((id, index) => {
+    const anchor = anchors.find((each) => each.id === id)!
+    return { anchorId: id, label: anchor.title || `Element ${index + 1}`, lines: anchor.source.split('\n') }
+  })
+}

+export interface Change {
+  anchorId: string
+  label: string
+  line: number
+  before: string
+  after: string
+  /** Where the name sits in `before`, for the highlight. */
+  marks: Token[]
+}
+
+/**
+ * The lines a rename of `from`, as defined at `at`, would change: that line's
+ * own name (unless it already holds the new one), then every use below, down
+ * to the next line that assigns `from` again.
+ */
+function planRename(
+  sheet: readonly SheetElement[],
+  at: { anchorId: string; line: number },
+  from: string,
+  to: string,
+  withDefinition: boolean,
+): Change[] {
+  const changes: Change[] = []
+  const swap = (element: SheetElement, line: number, tokens: Token[]) => {
+    if (tokens.length === 0) return
+    const before = element.lines[line]
+    let after = before
+    for (const token of [...tokens].reverse()) after = after.slice(0, token.start) + to + after.slice(token.end)
+    changes.push({ anchorId: element.anchorId, label: element.label, line, before, after, marks: tokens })
+  }
+  let reached = false
+  for (const element of sheet) {
+    for (let line = 0; line < element.lines.length; line++) {
+      const tokens = names(element.lines[line]).filter((token) => token.name === from)
+      const defined = definition(element.lines[line])
+      if (!reached) {
+        if (element.anchorId !== at.anchorId || line !== at.line) continue
+        reached = true
+        if (withDefinition) swap(element, line, tokens.slice(0, 1))
+        continue
+      }
+      if (defined?.params.includes(from)) continue
+      if (defined?.name === from) {
+        // Assigned again: its right-hand side still reads the old value, then the rename ends.
+        swap(element, line, tokens.slice(1))
+        return changes
+      }
+      swap(element, line, tokens)
+    }
+  }
+  return changes
+}
+
+/** Where `name`, read at `at`, was assigned: the nearest assignment at or above it. */
+function definedAt(sheet: readonly SheetElement[], at: { anchorId: string; line: number }, name: string) {
+  let found: { anchorId: string; line: number } | null = null
+  for (const element of sheet) {
+    for (let line = 0; line < element.lines.length; line++) {
+      if (definition(element.lines[line])?.name === name) found = { anchorId: element.anchorId, line }
+      if (element.anchorId === at.anchorId && line === at.line) return found
+    }
+  }
+  return found
+}
+
+/** Every changed element in one scene write: one undo step. */
+function writeChanges(api: ExcalidrawImperativeAPI, changes: readonly Change[]) {
+  const sources = new Map<string, string[]>()
+  for (const element of readSheet(api)) sources.set(element.anchorId, element.lines)
+  for (const change of changes) sources.get(change.anchorId)![change.line] = change.after
+  const touched = new Set(changes.map((change) => change.anchorId))
+  api.updateScene({
+    elements: api.getSceneElementsIncludingDeleted().map((element) =>
+      touched.has(element.id)
+        ? newElementWith(element, { customData: { ...element.customData, source: sources.get(element.id)!.join('\n') } })
+        : element,
+    ),
+    captureUpdate: CaptureUpdateAction.IMMEDIATELY,
+  })
+}
+
+function refusal(sheet: readonly SheetElement[], from: string, to: string): string | null {
+  if (to === '') return 'Type the new name.'
+  if (to === from) return null
+  if (!/^[A-Za-z_$À-ʯͰ-Ͽ][A-Za-z0-9_$À-ʯͰ-Ͽ]*$/.test(to)) {
+    return 'A name is letters, digits and _, and starts with a letter.'
+  }
+  if (sheet.some((element) => element.lines.some((line) => definition(line)?.name === to))) {
+    return `${to} is already a name on this sheet.`
+  }
+  return null
+}

+// ───────────────────────────── the prototype's state ─────────────────────────────
+
+interface Asking {
+  anchorId: string
+  line: number
+  from: string
+  /** Variant A knows the new name already; B and C ask for it. */
+  to: string | null
+  where: 'prompt' | 'panel' | 'line'
+  /** Variant C: the line F2 was pressed on, where its popover shows. */
+  near?: { anchorId: string; line: number }
+}
+
+interface Done {
+  from: string
+  to: string
+  changes: Change[]
+  steps: string
+}
+
+interface State {
+  api: ExcalidrawImperativeAPI | null
+  asking: Asking | null
+  done: Done | null
+}
+
+let state: State = { api: null, asking: null, done: null }
+const listeners = new Set<() => void>()
+function set(next: Partial<State>) {
+  state = { ...state, ...next }
+  listeners.forEach((listener) => listener())
+}
+function useProto(): State {
+  return useSyncExternalStore(
+    (listener) => {
+      listeners.add(listener)
+      return () => listeners.delete(listener)
+    },
+    () => state,
+  )
+}
+
+function rename(asking: Asking, to: string, steps: string, after?: () => void) {
+  const api = state.api
+  if (!api) return
+  const changes = planRename(readSheet(api), asking, asking.from, to, asking.where !== 'prompt')
+  writeChanges(api, changes)
+  set({ asking: null, done: { from: asking.from, to, changes, steps } })
+  api.setToast({
+    message: `Renamed ${asking.from} to ${to} in ${count(changes.length)}. Ctrl+Z takes it back.`,
+    closable: true,
+    duration: 6000,

+const count = (lines: number) => (lines === 1 ? '1 line' : `${lines} lines`)

+/** Variant A's hook, called by the overlay at every line commit (PROTOTYPE). */
+export function protoLineCommitted(anchorId: string, before: string, after: string) {
+  if (renameVariant() !== 'A' || !state.api) return
+  const was = before.split('\n')
+  const now = after.split('\n')
+  if (was.length !== now.length) return
+  const changed = now.flatMap((text, line) => (text === was[line] ? [] : [line]))
+  if (changed.length !== 1) return
+  const line = changed[0]
+  const from = definition(was[line])?.name
+  const to = definition(now[line])?.name
+  if (!from || !to || from === to) return
+  // The scene takes the commit in this same turn; the prompt reads it after.
+  window.setTimeout(() => {
+    const api = state.api
+    if (!api) return
+    const asking: Asking = { anchorId, line, from, to, where: 'prompt' }
+    if (planRename(readSheet(api), asking, from, to, false).length > 0) set({ asking })
+  })
+}

+/** The field for the new name, the lines that change, Rename and Cancel (B and C). */
+function RenameForm({ asking, steps, onDone }: { asking: Asking; steps: string; onDone?: () => void }) {
+  const api = state.api!
+  const [to, setTo] = useState(asking.from)
+  const field = useRef<HTMLInputElement>(null)
+  useEffect(() => {
+    field.current?.focus()
+    field.current?.select()
+  }, [])
+  const sheet = readSheet(api)
+  const typed = to.trim()
+  const problem = refusal(sheet, asking.from, typed)
+  const changes = planRename(sheet, asking, asking.from, typed === '' ? asking.from : typed, true)
+  const ready = problem === null && typed !== asking.from
+  const cancel = () => {
+    set({ asking: null })
+    onDone?.()
+  }
+  return (
+    <form
+      className="rename-proto__form"
+      onSubmit={(event) => {
+        event.preventDefault()
+        if (ready) rename(asking, typed, steps, onDone)
+      }}
+      onKeyDown={(event) => {
+        event.stopPropagation()
+        if (event.key === 'Escape') cancel()
+      }}
+    >
+      <label className="rename-proto__label">
+        <span>
+          Rename <code>{asking.from}</code> to
+        </span>
+        <input
+          ref={field}
+          className="rename-proto__field"
+          type="text"
+          value={to}
+          spellCheck={false}
+          autoComplete="off"
+          aria-label="New name"
+          onChange={(event) => setTo(event.target.value)}
+        />
+      </label>
+      {problem ? (
+        <p className="rename-proto__problem">{problem}</p>
+      ) : (
+        <p className="rename-proto__count">Changes {count(changes.length)}:</p>
+      )}
+      <ChangeList changes={changes} />
+      <div className="rename-proto__buttons">
+        <button type="submit" className="rename-proto__primary" disabled={!ready}>
+          Rename
+        </button>
+        <button type="button" onClick={cancel}>
+          Cancel
+        </button>
+        <span className="rename-proto__keys">Enter · Esc</span>
+      </div>
+    </form>
+  )
+}

+function ChangeList({ changes }: { changes: readonly Change[] }) {
+  return (
+    <ul className="rename-proto__changes">
+      {changes.map((change) => (
+        <li key={`${change.anchorId}:${change.line}`}>
+          <span className="rename-proto__where">
+            {change.label}, line {change.line + 1}
+          </span>
+          <code>{marked(change)}</code>
+        </li>
+      ))}
+    </ul>
+  )
+}
+
+function marked(change: Change) {
+  const parts: (string | React.JSX.Element)[] = []
+  let from = 0
+  for (const mark of change.marks) {
+    parts.push(change.before.slice(from, mark.start), <mark key={mark.start}>{mark.name}</mark>)
+    from = mark.end
+  }
+  parts.push(change.before.slice(from))
+  return parts
+}

+interface LayerProps {
+  api: ExcalidrawImperativeAPI
+  leaveLine: () => void
+  focusCanvas: () => void

+/** Variant A's prompt, variant C's popover, the last rename and the switcher (PROTOTYPE). */
+export function RenamePrototype({ api, leaveLine, focusCanvas }: LayerProps) {
+  const variant = renameVariant()
+  const { asking, done } = useProto()
+  useLayoutEffect(() => {
+    set({ api })
+    return () => set({ api: null, asking: null })
+  }, [api])
+
+  // Variant C: F2 in the open line asks about the name at the caret, else the line's own.
+  useEffect(() => {
+    if (variant !== 'C') return
+    const onKey = (event: KeyboardEvent) => {
+      if (event.key !== 'F2') return
+      const field = document.activeElement
+      if (!(field instanceof HTMLTextAreaElement) || !field.classList.contains('math-element__field')) return
+      const anchorId = field.closest<HTMLElement>('[data-anchor-id]')?.dataset.anchorId
+      const row = field.closest<HTMLElement>('[data-line]')?.dataset.line
+      if (!anchorId || row === undefined) return
+      event.preventDefault()
+      event.stopPropagation()
+      const line = Number(row)
+      const caret = field.selectionStart
+      const at = names(field.value).find((token) => token.start <= caret && caret <= token.end)
+      const name = at?.name ?? definition(field.value)?.name
+      leaveLine()
+      const where = name ? definedAt(readSheet(api), { anchorId, line }, name) : null
+      if (!name || !where) {
+        api.setToast({
+          message: name ? `${name} isn't a name this sheet assigns.` : 'Put the caret on a name to rename it.',
+          closable: true,
+          duration: 4000,
+        })
+        focusCanvas()
+        return
+      }
+      set({ asking: { ...where, from: name, to: null, where: 'line', near: { anchorId, line } } })
+    }
+    document.addEventListener('keydown', onKey, true)
+    return () => document.removeEventListener('keydown', onKey, true)
+  }, [variant, api, leaveLine, focusCanvas])
+
+  // Variant A's prompt goes when the sheet no longer needs it (an undo, an edit of a use).
+  useEffect(() => {
+    if (asking?.where !== 'prompt') return
+    return api.onChange(() => {
+      const still = state.asking
+      if (still?.where !== 'prompt' || still.to === null) return
+      if (planRename(readSheet(api), still, still.from, still.to, false).length === 0) set({ asking: null })
+    })
+  }, [api, asking])
+
+  if (!variant) return null
+  const floating = asking && asking.where !== 'panel' ? asking : null
+  return (
+    <div className="rename-proto">
+      {floating && (
+        <Beside
+          api={api}
+          anchorId={(floating.near ?? floating).anchorId}
+          line={(floating.near ?? floating).line}
+          below={floating.where === 'line'}
+        >
+          {floating.where === 'prompt' ? (
+            <Prompt asking={floating} leaveLine={leaveLine} />
+          ) : (
+            <RenameForm
+              asking={floating}
+              steps="1 undo step: Ctrl+Z puts every line back at once."
+              onDone={focusCanvas}
+            />
+          )}
+        </Beside>
+      )}
+      {done && <LastRename done={done} />}
+      <Switcher variant={variant} />
+    </div>
+  )
+}

+/** Variant A: "Rename rho to rho_w in 7 lines?", Rename and Keep. */
+function Prompt({ asking, leaveLine }: { asking: Asking; leaveLine: () => void }) {
+  const api = state.api!
+  const [listed, setListed] = useState(false)
+  const changes = planRename(readSheet(api), asking, asking.from, asking.to!, false)
+  return (
+    <div className="rename-proto__form is-prompt" role="dialog" aria-label="Rename">
+      <p className="rename-proto__question">
+        Rename <code>{asking.from}</code> to <code>{asking.to}</code> in{' '}
+        <button type="button" className="rename-proto__link" onMouseDown={(e) => e.preventDefault()} onClick={() => setListed(!listed)}>
+          {count(changes.length)}
+        </button>
+        ?
+      </p>
+      {listed && <ChangeList changes={changes} />}
+      <div className="rename-proto__buttons">
+        <button
+          type="button"
+          className="rename-proto__primary"
+          onMouseDown={(event) => event.preventDefault()}
+          onClick={() => {
+            // A line opened since (↓ or Enter) is written first, as its own step.
+            leaveLine()
+            rename(
+              asking,
+              asking.to!,
+              '2 undo steps: the first Ctrl+Z puts the uses back (they read the old name again, so they show errors), the second puts the edited line back.',
+            )
+          }}
+        >
+          Rename
+        </button>
+        <button type="button" onMouseDown={(event) => event.preventDefault()} onClick={() => set({ asking: null })}>
+          Keep {asking.from}
+        </button>
+      </div>
+    </div>
+  )
+}

+/** Places its children beside a line of the sheet, and follows it. */
+function Beside({
+  api,
+  anchorId,
+  line,
+  below,
+  children,
+}: {
+  api: ExcalidrawImperativeAPI
+  anchorId: string
+  line: number
+  below: boolean
+  children: React.ReactNode
+}) {
+  const box = useRef<HTMLDivElement>(null)
+  useLayoutEffect(() => {
+    let frame = 0
+    const place = () => {
+      const element = document.querySelector<HTMLElement>(`[data-anchor-id="${anchorId}"]`)
+      const row = element?.querySelector<HTMLElement>(`[data-line="${line}"]`)
+      const stage = box.current?.offsetParent?.getBoundingClientRect()
+      if (box.current && element && row && stage) {
+        const at = row.getBoundingClientRect()
+        const whole = element.getBoundingClientRect()
+        const sidebar = document.querySelector('.excalidraw .sidebar')?.getBoundingClientRect()
+        const edge = (sidebar && sidebar.width > 0 ? sidebar.left : stage.right) - box.current.offsetWidth - 12
+        const left = below ? at.left : whole.right + 10
+        box.current.style.left = `${Math.max(8, Math.min(left, edge)) - stage.left}px`
+        box.current.style.top = `${(below ? at.bottom + 6 : at.top - 6) - stage.top}px`
+      }
+      frame = requestAnimationFrame(place)
+    }
+    place()
+    return () => cancelAnimationFrame(frame)
+  }, [api, anchorId, line, below])
+  return (
+    <div ref={box} className="rename-proto__beside">
+      {children}
+    </div>
+  )
+}

+function Switcher({ variant }: { variant: RenameVariant }) {
+  const keys = Object.keys(VARIANTS) as RenameVariant[]
+  const go = (by: number) => {
+    const next = keys[(keys.indexOf(variant) + by + keys.length) % keys.length]
+    const url = new URL(window.location.href)
+    url.searchParams.set('rename', next)
+    window.location.assign(url)
+  }
+  return (
+    <nav className="rename-proto__switcher" aria-label="Prototype variants">
+      <button type="button" aria-label="Previous variant" onClick={() => go(-1)}>
+        ←
+      </button>
+      <span>
+        {variant} — {VARIANTS[variant]}
+      </span>
+      <button type="button" aria-label="Next variant" onClick={() => go(1)}>
+        →
+      </button>
+    </nav>
+  )
