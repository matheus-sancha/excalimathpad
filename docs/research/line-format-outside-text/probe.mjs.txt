// Part 1: a one-line <input> and newlines; a textarea's undo.
import { createRequire } from 'module'
const require = createRequire('/home/user/excalimathpad/')
let chromium
try { ({ chromium } = require('playwright')) } catch { ({ chromium } = createRequire(import.meta.url)('playwright')) }
const b = await chromium.launch()
const p = await b.newPage()
await p.setContent(`<input id=i style="font:16px monospace;width:200px"><textarea id=t rows=1 style="font:16px monospace;width:200px;resize:none;overflow:hidden;white-space:pre-wrap"></textarea>`)
const out = await p.evaluate(() => {
  const i = document.getElementById('i'), t = document.getElementById('t')
  i.focus(); document.execCommand('insertText', false, 'a = 1\nb = 2')
  const typedInput = JSON.stringify(i.value)
  i.value = 'x\ny'; const setInput = JSON.stringify(i.value)
  t.focus(); document.execCommand('insertText', false, 'check = if(TP <= T_p, "Pass", "Fail") @text-green if result == "Pass"')
  const before = t.value.length
  document.execCommand('undo')
  return { typedInput, setInput, textareaScrollHeight: t.scrollHeight, clientHeight: t.clientHeight, textareaUndoWorks: t.value.length < before, selStart: typeof t.selectionStart }
})
console.log(JSON.stringify(out))
await b.close()
