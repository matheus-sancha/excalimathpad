+ * PROTOTYPE, throwaway (wayfinder ticket #181). Four variants of the Sheet
+ * panel's width, resizing, docking and opening on an edit, in the real panel,
+ * switched by `?panel=A|B|C|D` (and `&edit=always|docked|never` to override
+ * when an edit opens the panel). Not production code: no tests, no memory

+import { createMathAnchorElements } from '../canvas/mathAnchor.ts'

+type Edit = 'always' | 'docked' | 'never'
+interface Variant {
+  key: string
+  name: string
+  /** The panel's width for an editor of this width. */
+  width: (editor: number, dragged: number) => number
+  resize: boolean
+  iconTabs: boolean
+  /** The editor docks the panel when wider than this. */
+  docksAbove: (width: number) => number
+  /** Below the docking width: float over the sheet, or the bottom sheet. */
+  narrow: 'float' | 'bottom'
+  edit: Edit
+}

+const TOOLBAR_ROOM = 640
+const VARIANTS: Variant[] = [

+    key: 'A',
+    name: 'Today: 384, docks above 1229',
+    width: () => 384,
+    resize: false,
+    iconTabs: false,

+    narrow: 'float',
+    edit: 'always',
+  },

+    name: 'Slim: 320 fixed, docks above 960',
+    width: () => 320,

+    edit: 'docked',

+/** Live state, kept outside React so App, SheetCanvas and the bar read one value. */

+const listeners = new Set<() => void>()
+function set(next: Partial<typeof state>) {

+const subscribe = (l: () => void) => (listeners.add(l), () => void listeners.delete(l))

+export const panelProto = {

+/** Re-renders on a width or drag change and keeps the CSS variable in step. */
+export function usePanelProto(root: HTMLElement | null) {
+  const snapshot = useSyncExternalStore(subscribe, () => state)
+  useEffect(() => {
+    if (!variant || !root) return
+    const measure = () => set({ editor: root.clientWidth })
+    measure()
+    const observer = new ResizeObserver(measure)
+    observer.observe(root)
+    return () => observer.disconnect()
+  }, [root])
+  const width = panelProto.width()
+  useEffect(() => {
+    if (!variant) return
+    document.documentElement.style.setProperty('--sheet-panel-width', `${width}px`)
+    document.documentElement.classList.toggle('proto-icon-tabs', variant.iconTabs)
+  }, [width])
+  return snapshot
