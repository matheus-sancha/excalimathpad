// A probe of docs/research/chart-spill.md (wayfinder ticket #176), made from scripts/touch-run.mjs, whose header follows.
// To run it: copy to scripts/ as .mjs, npm run build:plugin, then OUT=<a directory> node scripts/<name>.mjs

// The touch run (docs/spec-v1.md section 4, Touch; section 8, On a phone;
// section 12a, Releases; section 14, Two taps reach the app first): the built
// `dist-plugin/main.js` in headless Chromium against the fake `obsidian`, with
// touch (Playwright's `hasTouch`) at a phone and a tablet viewport. Each runs
// twice: as Android, and as iOS, where Excalidraw cancels every `touchstart`
// on its canvas, so the browser sends no click (`navigator.platform` is what
// Excalidraw reads). On the phone the plugin's `Platform.isPhone` is set, so
// the Sheet panel is the bottom sheet.
//
// It covers what the device checklist can't leave to chance: a double tap
// opens a line (and a drawing on top keeps Excalidraw's own), the Format tab's
// Done, ↑ ↓ and Edit, the canvas hint by tap and by click, a tap's tips and the slots and ⚙ a touch selection
// shows, the quick keys and their undo, the panel's open and restore, and the
// suggestions opening above the field; on the tablet, a mouse keeps hover. It
// runs in Chromium, so it covers neither WebKit nor a real on-screen keyboard:
// those are the device checklist's (CONTRIBUTING.md). It fails on any check
// that doesn't hold, and on any page error.
//
// Run: `node scripts/touch-run.mjs` (`npm run test:touch`), after
// `npm run build:plugin`.

import { build } from 'esbuild'
import { createServer } from 'node:http'
import { readFile } from 'node:fs/promises'
import { createRequire } from 'node:module'
import { dirname, join } from 'node:path'
import { fileURLToPath } from 'node:url'

const root = join(dirname(fileURLToPath(import.meta.url)), '..')
const dist = join(root, 'dist-plugin')
const FILE = 'Calcs/touch.excalimathpad'

const fake = await build({
  absWorkingDir: root,
  entryPoints: ['plugin/test/obsidian.ts'],
  bundle: true,
  format: 'iife',
  globalName: 'FakeObsidian',
  write: false,
  target: ['es2022'],
})

const page = `<!doctype html>
<html lang="en"><head><meta charset="utf-8">
<meta name="viewport" content="width=device-width, initial-scale=1">
<link rel="stylesheet" href="/styles.css">
<style>html,body{margin:0;height:100%}.workspace,.workspace-leaf,.workspace-leaf-content,.view-content{height:100%}</style>
</head><body>
<script src="/fake-obsidian.js"></script>
<script>
  var module = { exports: {} }, exports = module.exports
  window.require = (name) => {
    if (name === 'obsidian') return FakeObsidian
    throw new Error('main.js required ' + name)
  }
</script>
<script src="/main.js"></script>
<script>
  window.__start = async (sheet, manifest, phone) => {
    FakeObsidian.Platform.isPhone = phone
    FakeObsidian.Platform.isTablet = !phone
    FakeObsidian.Platform.isMobile = true
    const app = new FakeObsidian.App()
    app.vault.add(${JSON.stringify(FILE)}, sheet)
    const plugin = new module.exports.default(app, manifest)
    await plugin.onload()
    const leaf = app.workspace.getLeaf()
    await leaf.openFile(new FakeObsidian.TFile(${JSON.stringify(FILE)}))
    window.__leaf = leaf
  }
  // Whether Excalidraw's text editor has opened since the last reset: in
  // Chromium's touch, as Android, it closes again by itself (the browser's
  // \`dblclick\` follows Excalidraw's double tap), the app or not.
  window.__editorSeen = false
  new MutationObserver(() => {
    if (document.querySelector('textarea.excalidraw-wysiwyg')) window.__editorSeen = true
  }).observe(document, { childList: true, subtree: true })
  // The scene's text elements, from the view's file as the app holds it.
  window.__texts = () => JSON.parse(window.__leaf.view.getViewData()).elements.filter((e) => !e.isDeleted && e.type === 'text').length
</script>
</body></html>`

const types = { '.js': 'text/javascript', '.css': 'text/css', '.json': 'application/json' }
const server = createServer(async (request, response) => {
  const path = new URL(request.url, 'http://127.0.0.1').pathname
  try {
    let body
    if (path === '/') body = page
    else if (path === '/fake-obsidian.js') body = fake.outputFiles[0].text
    else if (['/main.js', '/styles.css', '/manifest.json'].includes(path)) body = await readFile(join(dist, path))
    else throw new Error('not found')
    response.writeHead(200, { 'content-type': types[path.slice(path.lastIndexOf('.'))] ?? 'text/html' })
    response.end(body)
  } catch {
    response.writeHead(404)
    response.end()
  }
})
await new Promise((resolve) => server.listen(0, '127.0.0.1', resolve))
const origin = `http://127.0.0.1:${server.address().port}`

import { writeFileSync, mkdirSync } from 'node:fs'

const SOURCE = `# True position, RFS (ASME Y14.5)
dx = 0.08 mm
dy = -0.05 mm
T_p = 0.25 mm
TP = 2 sqrt(dx^2 + dy^2) in mm
check = if(TP <= T_p, "Pass", "Fail") @text-green if result == "Pass" @text-red if result == "Fail"
used = TP / T_p * 100
margin = T_p - TP in mm
# Graph: tolerance zone and the measured point
r_t = T_p / 2 in mm
theta = 0 .. 2 pi step pi / 90
zone_x = r_t cos(theta) in mm
zone(a) = r_t sin(a)
measured(a) = dy + 0 mm * a
plot(zone_x in mm, zone(theta) in mm, measured(theta) in mm), mark dx, x -2.3 r_t .. 2.3 r_t, y -1.2 r_t .. 1.2 r_t, size L, title "True position"`
const PLOT = SOURCE.split('\n').at(-1)
const el = (id, x, y, source, extra = {}) => ({ id, type: 'rectangle', x, y, width: 160, height: 60, customData: { kind: 'math', source, ...extra } })
const mk = (elements) => JSON.stringify({ type: 'excalidraw', version: 2, elements, appState: {}, files: {}, excalimathpad: { format: 1 } })
const { chromium } = createRequire(import.meta.url)('playwright')
const browser = await chromium.launch()
const manifest = JSON.parse(await readFile(join(dist, 'manifest.json'), 'utf8'))
const report = {}
const state = (tab) => tab.evaluate(() => {
  const R = (e) => { if (!e) return null; const r = e.getBoundingClientRect(); return { l: Math.round(r.left), t: Math.round(r.top), r: Math.round(r.right), b: Math.round(r.bottom) } }
  const body = document.querySelector('[data-anchor-id="calc"] .math-element__body')
  const scrolled = []
  for (let e = body; e; e = e.parentElement) if (e.scrollLeft || e.scrollTop) scrolled.push({ el: (e.className && String(e.className).slice(0, 60)) || e.tagName, scrollLeft: e.scrollLeft, scrollTop: e.scrollTop, overflow: getComputedStyle(e).overflow })
  const f = document.querySelector('.math-element__field')
  return { body: R(body), svg: R(document.querySelector('[data-anchor-id="calc"] svg')), field: R(f), caret: f && f.selectionStart, fieldLen: f && f.value.length, scrolled, overlay: R(document.querySelector('.math-overlay')), win: [window.scrollX, window.scrollY] }
})
const open = async (name, sheet, viewport, act) => {
  const context = await browser.newContext({ viewport, deviceScaleFactor: 1 })
  const tab = await context.newPage()
  await tab.goto(`${origin}/`)
  await tab.evaluate(([text, m]) => window.__start(text, m, false), [sheet, manifest])
  await tab.evaluate(() => { FakeObsidian.Platform.isMobile = false; FakeObsidian.Platform.isTablet = false })
  await tab.waitForSelector('[data-anchor-id="calc"] .math-element__plot-chart svg', { timeout: 30_000 })
  await tab.waitForTimeout(1000)
  report[name] = {}
  const snap = async (step) => { report[name][step] = await state(tab); await tab.screenshot({ path: join(OUT, `${name}-${step}.png`) }) }
  await act(tab, snap)
  await context.close()
}
const full = mk([el('calc', 40, 90, SOURCE)])
const base = SOURCE.split('\n').slice(0, -1).join('\n')
const short = mk([el('calc', 40, 90, base + '\nplot(zone_x, zone(theta))')])
const typeIt = (fix) => async (tab, snap) => {
  if (fix) await tab.addStyleTag({ content: '.app-shell__overlay{overflow:clip !important}' })
  const p = await tab.locator('[data-anchor-id="calc"] [data-line="14"]').boundingBox()
  await tab.mouse.dblclick(p.x + 30, p.y + 10); await tab.waitForTimeout(600)
  await tab.keyboard.press('ControlOrMeta+a')
  await tab.keyboard.type(PLOT, { delay: 5 }); await tab.waitForTimeout(600)
  await snap('1-typed')
  await tab.keyboard.press('Enter'); await tab.waitForTimeout(800)
  await snap('2-enter')
  await tab.keyboard.press('Escape'); await tab.waitForTimeout(800)
  await snap('3-escape')
  // does a canvas pan or zoom put it back?
  await tab.mouse.move(300, 500); await tab.mouse.wheel(0, 100); await tab.waitForTimeout(500)
  await snap('4-wheel')
}
await open('clip', short, { width: 1366, height: 768 }, typeIt(true))
await open('noclip', short, { width: 1366, height: 768 }, typeIt(false))
// chart ink against the chart's box and the element's, in several states
const ink = (tab) => tab.evaluate(() => {
  const R = (e) => { const r = e.getBoundingClientRect(); return { l: +r.left.toFixed(1), t: +r.top.toFixed(1), r: +r.right.toFixed(1), b: +r.bottom.toFixed(1) } }
  const wrap = document.querySelector('[data-anchor-id="calc"]')
  const body = R(wrap.querySelector('.math-element__body'))
  const svg = wrap.querySelector('.math-element__plot-chart svg'); const sv = R(svg)
  const z = (sv.r - sv.l) / svg.width.baseVal.value
  const past = [...svg.querySelectorAll('text')].map((t) => ({ s: t.textContent.slice(0, 50), ...R(t) })).map((t) => ({ s: t.s, pastSvg: { l: sv.l - t.l, r: t.r - sv.r, t: sv.t - t.t, b: t.b - sv.b }, pastBody: { l: body.l - t.l, r: t.r - body.r, t: body.t - t.t, b: t.b - body.b } }))
    .map((t) => ({ s: t.s, pastSvg: Object.fromEntries(Object.entries(t.pastSvg).filter(([, v]) => v > 0.5).map(([k, v]) => [k, +(v / z).toFixed(1)])), pastBody: Object.fromEntries(Object.entries(t.pastBody).filter(([, v]) => v > 0.5).map(([k, v]) => [k, +(v / z).toFixed(1)])) }))
    .filter((t) => Object.keys(t.pastSvg).length)
  return { zoom: +z.toFixed(3), bodyW: +((body.r - body.l) / z).toFixed(1), svgW: +((sv.r - sv.l) / z).toFixed(1), textPast: past, theme: document.querySelector('.excalidraw')?.className.match(/theme--\w+/)?.[0], bodyColor: getComputedStyle(svg).color }
})
