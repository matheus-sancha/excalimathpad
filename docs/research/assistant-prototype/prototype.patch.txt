+// ───────────────────────────── the prototype's own state ─────────────────────────────

+interface ProtoState {
+  /** B and C: the chat's own panel is open. */
+  open: boolean
+  hasKey: boolean
+  failNext: boolean
+}
+
+let state: ProtoState = {
+  open: true,
+  hasKey: typeof window === 'undefined' || new URLSearchParams(window.location.search).get('key') !== '0',
+  failNext: false,
+}
+const listeners = new Set<() => void>()
+function setProto(next: Partial<ProtoState>) {
+  state = { ...state, ...next }
+  listeners.forEach((listener) => listener())
+}
+function useProto(): ProtoState {
+  return useSyncExternalStore(
+    (listener) => {
+      listeners.add(listener)
+      return () => listeners.delete(listener)
+    },
+    () => state,
+  )
+}

+export function useAssistantPanel(): { open: boolean; toggle: () => void } {
+  const { open } = useProto()
+  return { open, toggle: () => setProto({ open: !state.open }) }

+let chat: Chat = { messages: [], waiting: false }
+const chatListeners = new Set<() => void>()
+function setChat(next: (now: Chat) => Chat) {
+  chat = next(chat)
+  chatListeners.forEach((listener) => listener())

+let timer: number | null = null
+function stopTimer() {
+  if (timer !== null) window.clearTimeout(timer)

+function write(answer: Answer) {
+  const total = tokens(answer).length
+  const step = () => {
+    let done = false
+    setChat((now) => {
+      const last = now.messages[now.messages.length - 1]
+      if (last?.from !== 'assistant' || last.state !== 'writing') return now
+      const shown = Math.min(total, last.shown + 2)
+      done = shown >= total
+      return { ...now, messages: [...now.messages.slice(0, -1), { ...last, shown, state: done ? 'done' : 'writing' }] }
+    })
+    timer = done ? null : window.setTimeout(step, 45)
+  }
+  step()
+}

+function ask(asked: string) {
+  const answer = answerFor(asked)
+  setChat((now) => ({ ...now, waiting: true }))
+  timer = window.setTimeout(() => {
+    const failed = state.failNext
+    if (failed) setProto({ failNext: false })
+    setChat((now) => ({
+      waiting: false,
+      messages: [...now.messages, { from: 'assistant', answer, shown: 0, state: failed ? 'failed' : 'writing' }],
+    }))
+    if (!failed) write(answer)
+  }, 1100)

+function stop() {
